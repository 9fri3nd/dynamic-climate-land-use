# --- Un-lagged solar declination (spec addendum 1; display path, lazy) ---
# delta = 0.409 * sin(2 * pi * (doy - 80) / 365) via odd Bhaskara on a = 2 * phase:
# sin(pi a) = 16 a (1 - |a|) / (4 (|a| - 0.5)^2 + 4); the phase is a generated month leaf.
dcl_solar_arg = {
    value = dcl_solar_month_phase
    multiply = 2
}

dcl_solar_abs_arg = {
    value = dcl_solar_arg
    abs = 0
}

dcl_solar_num = {
    value = 1
    subtract = dcl_solar_abs_arg
    multiply = dcl_solar_arg
    multiply = 16
}

dcl_solar_den = {
    value = dcl_solar_abs_arg
    subtract = 0.5
    pow = 2
    multiply = 4
    add = 4
}

calc_solar_declination = {
    value = dcl_solar_num
    divide = dcl_solar_den
    multiply = 0.409
}

# --- Lagged-declination helpers (spec addendum 2-3, feeding the temperature hot path) ---
dcl_lagged_declination_sq = {
    value = dcl_lagged_declination
    pow = 2
}

# tan(delta) ~= delta * (1 + delta^2 / 3)
dcl_tan_declination = {
    value = dcl_lagged_declination_sq
    multiply = 0.33333
    add = 1
    multiply = dcl_lagged_declination
}

# cos(delta) ~= 1 - delta^2 / 2 + delta^4 / 24 as a completed square (one delta^2 read)
dcl_cos_declination = {
    value = dcl_lagged_declination_sq
    subtract = 5.99952
    pow = 2
    multiply = 0.04167
    add = -0.49988
}

# Sunset hour-angle cosine x = clamp(-tan(lat) * tan(delta), -1, 1); y = x^2 (pow = 2)
dcl_sunset_cos_arg = {
    value = dcl_lat_tan
    multiply = dcl_tan_declination
    multiply = -1
    max = 1
    min = -1
}

dcl_sunset_cos_arg_sq = {
    value = dcl_sunset_cos_arg
    pow = 2
}

# omega_s / pi = 0.5 - x * (d1 + y * (d7 * (y - dq_s)^2 + dq_o)), d_i = c_i / pi.
# d3 + d5 y + d7 y^2 is run as the completed square d7 (y - dq_s)^2 + dq_o.

    value = dcl_sunset_cos_arg_sq
    subtract = 0.75749
    pow = 2
    multiply = 1.17402
    add = 0.19218
}

    value = dcl_sunset_cos_arg_sq

    add = 0.22627
}
