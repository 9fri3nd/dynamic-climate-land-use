# Thermal lag in days, 25 + (1 - continentality) * 15 (spec addendum 3).
# Public API: no in-mod consumer (the lagged-declination table bakes it in); kept for display.
dcl_thermal_lag_days = {
    value = 1
    subtract = dcl_continentality
    multiply = 15
    add = 25
}

# Area-level anomalies (written by the monthly pass); 0 until the area variable exists.
dcl_temp_anomaly = {
    value = 0
    if = {
        limit = { area ?= { has_variable = macro_temp_anomaly } }
        value = area.var:macro_temp_anomaly
    }
}

dcl_humidity_anomaly = {
    value = 0
    if = {
        limit = { area ?= { has_variable = macro_humidity_anomaly } }
        value = area.var:macro_humidity_anomaly
    }
}

dcl_location_front_type = {
    value = 0
    if = {
        limit = { area ?= { has_variable = dcl_front_type } }
        value = area.var:dcl_front_type
    }
    min = 0
    max = 3
}

# Temperature: T = T_off + G * (Q - 0.5 * cos(lat)) + H * decl^2 - lapse * elevation + anomaly.
dcl_half_lat_cos = {
    value = dcl_lat_cos
    multiply = 0.5
}

dcl_decl2_term = {
    value = dcl_lagged_declination_sq
    multiply = dcl_temp_decl2_gain
}

dcl_lapse_cooling = {
    value = dcl_elevation
    multiply = 0.0065
}

dynamic_location_temperature = {
    value = dcl_insolation
    subtract = dcl_half_lat_cos
    multiply = dcl_temp_gain
    add = dcl_temp_offset
    add = dcl_decl2_term
    subtract = dcl_lapse_cooling
    add = dcl_temp_anomaly
}

dcl_location_temperature_mean = {
    value = dcl_temp_offset
    subtract = dcl_lapse_cooling
}

dynamic_location_humidity = {
    value = dynamic_location_temperature
    subtract = dcl_location_temperature_mean
    multiply = -1.2
    add = dcl_humidity_base
    if = {
        limit = { is_coastal = yes }
        add = 20
    }
    add = dcl_humidity_anomaly
    min = 5
    max = 100
}
